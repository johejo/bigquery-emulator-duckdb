# Hints for //:function_probe.
#
#   query NAME SQL   probes SQL instead of the calls generated from the function's signatures;
#                    repeat it to cover each signature BigQuery documents
#
# NAME must be in tools/bigquery_functions.txt. Run
# `bazel run //:function_probe -- --signatures $PWD/tools/bigquery_functions.txt $PWD/tools/function_probe_hints.txt`

# Date and time arithmetic takes an INTERVAL expression, and a TIME has no DAY part.
query DATE_ADD SELECT DATE_ADD(DATE '2024-01-15', INTERVAL 1 DAY)
query DATE_SUB SELECT DATE_SUB(DATE '2024-01-15', INTERVAL 1 DAY)
query DATETIME_ADD SELECT DATETIME_ADD(DATETIME '2024-01-15 10:20:30', INTERVAL 1 DAY)
query DATETIME_SUB SELECT DATETIME_SUB(DATETIME '2024-01-15 10:20:30', INTERVAL 1 DAY)
query TIMESTAMP_ADD SELECT TIMESTAMP_ADD(TIMESTAMP '2024-01-15 10:20:30+00', INTERVAL 1 HOUR)
query TIMESTAMP_SUB SELECT TIMESTAMP_SUB(TIMESTAMP '2024-01-15 10:20:30+00', INTERVAL 1 HOUR)
query TIME_ADD SELECT TIME_ADD(TIME '10:20:30', INTERVAL 1 HOUR)
query TIME_SUB SELECT TIME_SUB(TIME '10:20:30', INTERVAL 1 HOUR)
query TIME_DIFF SELECT TIME_DIFF(TIME '10:20:30', TIME '09:00:00', MINUTE)
query TIME_TRUNC SELECT TIME_TRUNC(TIME '10:20:30', HOUR)
query GENERATE_TIMESTAMP_ARRAY SELECT GENERATE_TIMESTAMP_ARRAY(TIMESTAMP '2024-01-15 00:00:00+00', TIMESTAMP '2024-01-17 00:00:00+00', INTERVAL 1 DAY)

# Arguments whose valid values a type alone does not describe.
query ACOS SELECT ACOS(0.5)
query ASIN SELECT ASIN(0.5)
query BOOL SELECT BOOL(JSON 'true')
query INT64 SELECT INT64(JSON '2')
query FLOAT64 SELECT FLOAT64(JSON '1.5')
query FLOAT64 SELECT FLOAT64(JSON '1.5', wide_number_mode => 'round')
query DATETIME SELECT DATETIME(2024, 1, 15, 10, 20, 30)
query DATETIME SELECT DATETIME(DATE '2024-01-15', TIME '10:20:30')
query DATETIME SELECT DATETIME(TIMESTAMP '2024-01-15 10:20:30+00', 'Asia/Tokyo')
query DATETIME SELECT DATETIME(DATE '2024-01-15')
query DATETIME SELECT DATETIME('2024-01-15 10:20:30')
query TIMESTAMP SELECT TIMESTAMP('2024-01-15 10:20:30', 'Asia/Tokyo')
query TIMESTAMP SELECT TIMESTAMP(DATE '2024-01-15', 'Asia/Tokyo')
query TIMESTAMP SELECT TIMESTAMP(DATETIME '2024-01-15 10:20:30', 'Asia/Tokyo')
query STRING SELECT STRING(TIMESTAMP '2024-01-15 10:20:30+00', 'Asia/Tokyo')
query STRING SELECT STRING(JSON '"abc"')
query FROM_BASE64 SELECT FROM_BASE64('YWJj')
query PARSE_JSON SELECT PARSE_JSON('{"a": 1}')
query PARSE_JSON SELECT PARSE_JSON('1.5', wide_number_mode => 'round')
query ERROR SELECT IF(FALSE, ERROR('abc'), 1)
query FORMAT SELECT FORMAT('%d-%s', 2, 'abc')
query GROUPING SELECT x, GROUPING(x) FROM UNNEST([1, 2]) AS x GROUP BY ROLLUP(x)
query PERCENTILE_CONT SELECT PERCENTILE_CONT(x, 0.5) OVER () FROM UNNEST([1.5, 2.5]) AS x
query PERCENTILE_DISC SELECT PERCENTILE_DISC(x, 0.5) OVER () FROM UNNEST([1, 2]) AS x
query RANGE SELECT RANGE(DATE '2024-01-01', DATE '2024-02-01')

# Expressions with syntax of their own, which the analyzer does not resolve as function calls.
query ARRAY SELECT ARRAY(SELECT x FROM UNNEST([1, 2]) AS x)
query CASE SELECT CASE 2 WHEN 1 THEN 'a' ELSE 'b' END
query CASE SELECT CASE WHEN TRUE THEN 'a' ELSE 'b' END
query CAST SELECT CAST('2' AS INT64)
query CAST SELECT CAST('2024-01-15' AS DATE FORMAT 'YYYY-MM-DD')
query SAFE_CAST SELECT SAFE_CAST('x' AS INT64)
query EXTRACT SELECT EXTRACT(DAY FROM DATE '2024-01-15')
query EXTRACT SELECT EXTRACT(HOUR FROM DATETIME '2024-01-15 10:20:30')
query EXTRACT SELECT EXTRACT(DAY FROM INTERVAL 1 DAY)
query EXTRACT SELECT EXTRACT(HOUR FROM TIME '10:20:30')
query EXTRACT SELECT EXTRACT(HOUR FROM TIMESTAMP '2024-01-15 10:20:30+00' AT TIME ZONE 'Asia/Tokyo')
