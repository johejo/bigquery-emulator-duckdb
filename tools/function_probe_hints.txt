# Hints for //:function_probe.
#
#   query NAME SQL   probes SQL instead of the calls generated from the function's signatures;
#                    repeat it to cover each signature BigQuery documents
#   note NAME TEXT   shows TEXT in the Notes column, for behavior a probe cannot observe;
#                    at most one per function
#
# NAME must be in tools/bigquery_functions.txt. Run
# `bazel run //:function_probe -- --signatures $PWD/tools/bigquery_functions.txt $PWD/tools/function_probe_hints.txt`

# Date and time arithmetic takes an INTERVAL expression, and a TIME has no DAY part.
query DATE_ADD SELECT DATE_ADD(DATE '2024-01-15', INTERVAL 1 DAY)
query DATE_SUB SELECT DATE_SUB(DATE '2024-01-15', INTERVAL 1 DAY)
query DATETIME_ADD SELECT DATETIME_ADD(DATETIME '2024-01-15 10:20:30', INTERVAL 1 DAY)
query DATETIME_SUB SELECT DATETIME_SUB(DATETIME '2024-01-15 10:20:30', INTERVAL 1 DAY)
query TIMESTAMP_ADD SELECT TIMESTAMP_ADD(TIMESTAMP '2024-01-15 10:20:30+00', INTERVAL 1 HOUR)
query TIMESTAMP_SUB SELECT TIMESTAMP_SUB(TIMESTAMP '2024-01-15 10:20:30+00', INTERVAL 1 HOUR)
query TIME_ADD SELECT TIME_ADD(TIME '10:20:30', INTERVAL 1 HOUR)
query TIME_SUB SELECT TIME_SUB(TIME '10:20:30', INTERVAL 1 HOUR)
query TIME_DIFF SELECT TIME_DIFF(TIME '10:20:30', TIME '09:00:00', MINUTE)
query TIME_TRUNC SELECT TIME_TRUNC(TIME '10:20:30', HOUR)
query GENERATE_TIMESTAMP_ARRAY SELECT GENERATE_TIMESTAMP_ARRAY(TIMESTAMP '2024-01-15 00:00:00+00', TIMESTAMP '2024-01-17 00:00:00+00', INTERVAL 1 DAY)

# Behavior that follows BigQuery where DuckDB would differ.
note BYTE_LENGTH counts the bytes of a STRING's UTF-8 encoding
note LENGTH counts characters of a STRING and bytes of a BYTES
note ARRAY_CONCAT returns NULL when any array is NULL
note SUBSTR starts at the first character for a position of 0 or before the start
note SUBSTRING starts at the first character for a position of 0 or before the start
note LPAD pads with spaces by default
note CONTAINS_SUBSTR compares NFKC normal forms, case folded; takes STRING only
note RPAD pads with spaces by default
note EXTRACT DAYOFWEEK and WEEK count from Sunday, ISOWEEK from Monday; sub-second parts, AT TIME ZONE and DATE / TIME / DATETIME parts follow BigQuery
note GENERATE_DATE_ARRAY takes only a DAY or WEEK step
note DATE_TRUNC WEEK starts on Sunday and ISOWEEK on Monday
note DATETIME_TRUNC WEEK starts on Sunday and ISOWEEK on Monday
note TIMESTAMP_TRUNC WEEK starts on Sunday and ISOWEEK on Monday
note DATE_DIFF WEEK starts on Sunday and ISOWEEK on Monday
note DATETIME_DIFF WEEK starts on Sunday and ISOWEEK on Monday; parts below a day count whole units rather than boundaries
note TIMESTAMP_DIFF counts whole units rather than boundaries
note TIME_DIFF counts whole units rather than boundaries
note INT64 from JSON, fails when the value has another JSON type
note FLOAT64 from JSON, fails when the value has another JSON type
note BOOL from JSON, fails when the value has another JSON type
note STRING from JSON, fails when the value has another JSON type
note FORMAT NUMERIC, JSON, ARRAY, STRUCT, INTERVAL and RANGE values are unsupported
