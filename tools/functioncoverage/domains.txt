# Reviewed expansions of prose type domains in the saved specification, not signatures discovered
# from the emulator. Each signature's function must have a section in its page.
#
#   page PAGE                          names the page under reference/standard-sql of later signatures
#   types @NAME = ITEM | ITEM...       names a list of types; an ITEM can use another list, such as
#                                      ARRAY<@NAME>, and -ITEM removes one
#   for $VAR in ITEM | ITEM... / end   repeats the lines in between with $VAR replaced by each type
#   NAME(ARGS) -> RESULT               a signature; [, ARG] is optional, `literal TYPE` keeps the
#                                      argument a literal, and TYPE = SAMPLE replaces its sample
#   note NAME TEXT                     documents what the function's expansion leaves out
#
# A function's signatures keep the order in which these lines produce them.

# ARRAY/STRUCT domains are necessarily representative: nested structural types have infinitely
# many shapes. Cover ordinary scalar families and nested fields; graph and MEASURE values need
# fixtures and are not included in these domains.
types @scalar = INT64 | NUMERIC | BIGNUMERIC | FLOAT64 | BOOL | STRING | BYTES | DATE | DATETIME | TIME | TIMESTAMP | INTERVAL | GEOGRAPHY | JSON | RANGE<DATE> | RANGE<DATETIME> | RANGE<TIMESTAMP>
types @struct = STRUCT<x INT64, label STRING> | STRUCT<n NUMERIC, nested STRUCT<flag BOOL>> | STRUCT<items ARRAY<BIGNUMERIC>, span RANGE<DATE>>
types @nonarray = @scalar | @struct
types @all = @nonarray | ARRAY<@nonarray>
# The specification's orderable-type exclusions.
types @orderable = @scalar | -GEOGRAPHY | -JSON
types @numeric = INT64 | NUMERIC | BIGNUMERIC | FLOAT64
types @text = STRING | BYTES
# The specification's comparable exclusions. STRUCTs compare only for equality.
# ARRAY isn't comparable, so neither is a STRUCT with an ARRAY field.
types @comparable = @nonarray | -GEOGRAPHY | -JSON | -STRUCT<items ARRAY<BIGNUMERIC>, span RANGE<DATE>>
types @range = RANGE<DATE> | RANGE<DATETIME> | RANGE<TIMESTAMP>
# The SQL types listed under the specification's JSON encodings.
types @json_encodable = @all | -GEOGRAPHY | -ARRAY<GEOGRAPHY>
