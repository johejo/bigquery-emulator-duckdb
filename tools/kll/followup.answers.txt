# Follow-up to maintainer answers archived in tools/kll/.
# All phi arguments are literal: BigQuery rejected the original per-row phi queries.
# Fixed bytes are copied from BigQuery observations, never emulator output.
# Do not replace these with INIT calls: compaction would change the sampled state.

# Rank selection, with the original four-element exact populations.
SELECT [KLL_QUANTILES.EXTRACT_POINT_INT64(s, 0.0), KLL_QUANTILES.EXTRACT_POINT_INT64(s, 0.249), KLL_QUANTILES.EXTRACT_POINT_INT64(s, 0.25), KLL_QUANTILES.EXTRACT_POINT_INT64(s, 0.251), KLL_QUANTILES.EXTRACT_POINT_INT64(s, 0.5), KLL_QUANTILES.EXTRACT_POINT_INT64(s, 0.75), KLL_QUANTILES.EXTRACT_POINT_INT64(s, 1.0)]
FROM (SELECT KLL_QUANTILES.INIT_INT64(x) AS s FROM UNNEST([10, 20, 30, 40]) AS x)
?> [{"v":"10"},{"v":"10"},{"v":"10"},{"v":"20"},{"v":"20"},{"v":"30"},{"v":"40"}]

SELECT [KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 0.0), KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 0.249), KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 0.25), KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 0.251), KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 0.5), KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 0.75), KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 1.0)]
FROM (SELECT KLL_QUANTILES.INIT_FLOAT64(x) AS s FROM UNNEST(ARRAY<FLOAT64>[10, 20, 30, 40]) AS x)
?> [{"v":"10.0"},{"v":"10.0"},{"v":"10.0"},{"v":"20.0"},{"v":"20.0"},{"v":"30.0"},{"v":"40.0"}]

# Compacted INT64 specimens, precision=1, inputs GENERATE_ARRAY(1,n).
# n=4: payload field 6 first appears in this recorded sequence.

# n=16: sum(level item counts * 2^level)=8, but outer n=16.

# Merge fixed compacted bytes; check exact extrema, not random interior quantiles.
SELECT STRUCT(KLL_QUANTILES.MERGE_POINT_INT64(s, 0.0) AS minimum, KLL_QUANTILES.MERGE_POINT_INT64(s, 1.0) AS maximum)
FROM UNNEST([
  FROM_HEX('0871100420048a071d080310011a01012201042a0212002a041202010232070a010410011801'),
  FROM_HEX('0871101020048a0724080310011a01012201102a0212002a0212002a0212002a0312010432070a011010071803')
]) AS s
?> {"f":[{"v":"1"},{"v":"16"}]}
