# Different precisions, NULL sketches, and all four non-BYTES MERGE functions.
WITH sketches AS (
  SELECT KLL_QUANTILES.INIT_INT64(x, 1000) AS s FROM UNNEST([1, 2]) AS x
  UNION ALL
  SELECT KLL_QUANTILES.INIT_INT64(x, 2000) FROM UNNEST([3, 4]) AS x
  UNION ALL SELECT CAST(NULL AS BYTES)
)
SELECT STRUCT(KLL_QUANTILES.MERGE_INT64(s, 4) AS quantiles, KLL_QUANTILES.MERGE_POINT_INT64(s, 0.5) AS point) FROM sketches
?> {"f":[{"v":[{"v":"1"},{"v":"1"},{"v":"2"},{"v":"3"},{"v":"4"}]},{"v":"2"}]}

WITH sketches AS (
  SELECT KLL_QUANTILES.INIT_FLOAT64(x, 1000) AS s FROM UNNEST(ARRAY<FLOAT64>[1, 2]) AS x
  UNION ALL
  SELECT KLL_QUANTILES.INIT_FLOAT64(x, 2000) FROM UNNEST(ARRAY<FLOAT64>[3, 4]) AS x
  UNION ALL SELECT CAST(NULL AS BYTES)
)
SELECT STRUCT(KLL_QUANTILES.MERGE_FLOAT64(s, 4) AS quantiles, KLL_QUANTILES.MERGE_POINT_FLOAT64(s, 0.5) AS point) FROM sketches
?> {"f":[{"v":[{"v":"1.0"},{"v":"1.0"},{"v":"2.0"},{"v":"3.0"},{"v":"4.0"}]},{"v":"2.0"}]}

# NaN retention, infinities, signed zero: serialize values as strings for inspection.
SELECT ARRAY(SELECT FORMAT('%T', v) FROM UNNEST(KLL_QUANTILES.EXTRACT_FLOAT64(s, 6)) AS v WITH OFFSET AS i ORDER BY i)
FROM (SELECT KLL_QUANTILES.INIT_FLOAT64(x) AS s FROM UNNEST([CAST('NaN' AS FLOAT64), CAST('-inf' AS FLOAT64), -1.0, CAST('-0.0' AS FLOAT64), 0.0, 1.0, CAST('inf' AS FLOAT64)]) AS x)
?> [{"v":"CAST(\"nan\" AS FLOAT64)"},{"v":"CAST(\"-inf\" AS FLOAT64)"},{"v":"-1.0"},{"v":"0.0"},{"v":"0.0"},{"v":"1.0"},{"v":"CAST(\"inf\" AS FLOAT64)"}]

SELECT FORMAT('%T', KLL_QUANTILES.EXTRACT_POINT_FLOAT64(s, 0.5))
FROM (SELECT KLL_QUANTILES.INIT_FLOAT64(x) AS s FROM UNNEST([CAST('NaN' AS FLOAT64), CAST('NaN' AS FLOAT64)]) AS x)
?> "CAST(\"nan\" AS FLOAT64)"
